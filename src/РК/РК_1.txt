-- Создаем схему (если не существует)
CREATE SCHEMA IF NOT EXISTS marketplace;

-- 1. Справочник клиентов
CREATE TABLE IF NOT EXISTS marketplace.clients (
    client_id      BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name           VARCHAR(100) NOT NULL,
    surname        VARCHAR(100) NOT NULL,
    phone_number   VARCHAR(20),
    created_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Справочник курьеров
CREATE TABLE IF NOT EXISTS marketplace.courier (
    courier_id     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name           VARCHAR(100) NOT NULL,
    surname        VARCHAR(100) NOT NULL,
    courier_type   VARCHAR(20),
    created_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. Справочник магазинов
CREATE TABLE IF NOT EXISTS marketplace.stores (
    store_id       BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    store_name     VARCHAR(150) NOT NULL,
    working_hours  VARCHAR(50),
    created_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 4. Справочник поставщиков
CREATE TABLE IF NOT EXISTS marketplace.suppliers (
    supplier_id    BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    supplier_name  VARCHAR(150) NOT NULL,
    created_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 5. Товары (зависит от suppliers)
CREATE TABLE IF NOT EXISTS marketplace.items (
    item_id        BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    item_desc      VARCHAR(255) NOT NULL,
    supplier_id    BIGINT NOT NULL,
    created_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_items_supplier FOREIGN KEY (supplier_id) REFERENCES marketplace.suppliers(supplier_id)
);

-- 6. Доставки (зависит от courier)
CREATE TABLE IF NOT EXISTS marketplace.delivery (
    delivery_id    BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    delivery_status VARCHAR(50) NOT NULL,
    courier_id     BIGINT NOT NULL,
    created_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_delivery_courier FOREIGN KEY (courier_id) REFERENCES marketplace.courier(courier_id)
);

-- 7. Заказы (зависит от clients, stores, delivery)
CREATE TABLE IF NOT EXISTS marketplace.orders (
    order_id       BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    delivery_id    BIGINT NOT NULL,
    store_id       BIGINT NOT NULL,
    client_id      BIGINT NOT NULL,
    created_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_orders_delivery FOREIGN KEY (delivery_id) REFERENCES marketplace.delivery(delivery_id),
    CONSTRAINT fk_orders_store FOREIGN KEY (store_id) REFERENCES marketplace.stores(store_id),
    CONSTRAINT fk_orders_client FOREIGN KEY (client_id) REFERENCES marketplace.clients(client_id)
);

-- 8. Детали заказов (зависит от orders, items)
CREATE TABLE IF NOT EXISTS marketplace.orders_details (
    order_id       BIGINT NOT NULL,
    order_line_id  BIGINT NOT NULL,
    item_id        BIGINT NOT NULL,
    qty            INT NOT NULL,
    "cost"         NUMERIC(10, 2) NOT NULL,
    created_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_dttm   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (order_id, order_line_id),
    CONSTRAINT fk_details_order FOREIGN KEY (order_id) REFERENCES marketplace.orders(order_id),
    CONSTRAINT fk_details_item FOREIGN KEY (item_id) REFERENCES marketplace.items(item_id)
);
